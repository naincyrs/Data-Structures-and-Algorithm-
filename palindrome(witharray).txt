FOR INTERGER; 

#include <bits/stdc++.h>
using namespace std;

int f(int arr[], int l, int r) {
    if (l >= r) {
        return true;
    }
    if(arr[l] != arr[r]){
        return false;
    }
    return f(arr, l + 1, r - 1); 
    
}

int main() {
    int n;
    cin >> n;
    int arr[n];

    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
    if (f(arr, 0, n - 1)) {
        cout << "true" << endl;
    } else {
        cout << "false" << endl;
    }

    return 0;
}


FOR STRING :

#include <bits/stdc++.h>
using namespace std;

int f(string &a, int l, int r) {
    if (l >= r) {
        return true;
    }
    if(a[l] != a[r]){
        return false;
    }
    return f(a, l + 1, r - 1); 
    
}

int main() {
    string a;
    cin >> a;
    
    if (f(a, 0, a.length() - 1)) {
        cout << "true" << endl;
    } else {
        cout << "false" << endl;
    }

    return 0;
}
