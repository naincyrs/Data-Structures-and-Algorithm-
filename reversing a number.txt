#include<bits/stdc++.h>
using namespace std;

int reverse(int n) {
    if (n == 0) return 0;  // Edge case
    // Handle negative numbers
       bool isNegative = false;
           if (n < 0) {
        isNegative = true;
        n = abs(n);
        }      
      int num=0 ;
       while (n>0) {
        int digit=n%10;
        num= num*10+digit;
        n=n/10;
    }
return isNegative ? -num : num; //ternary operator
}

int main() {
    int n;
    cout << "Enter an integer: ";
    cin >> n;

    int newnumber = reverse(n);
    cout << "Reversed number is: " << newnumber << endl;

    return 0;
}
