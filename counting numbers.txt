#include<bits/stdc++.h>
using namespace std;

int countDigit(int n) {
    if (n == 0) return 1;  // Edge case
    n = abs(n);            // Handle negative numbers
    int count = 0;

    while (n > 0) {
        n = n / 10;
        count++;
    }

    return count;
}

int main() {
    int n;
    cout << "Enter an integer: ";
    cin >> n;

    int digitCount = countDigit(n);
    cout << "Number of digits: " << digitCount << endl;

    return 0;
}
